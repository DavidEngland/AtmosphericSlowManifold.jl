module ModelSelection

using LinearAlgebra
using ..Discovery: DiscoveredModel

export ModelSelectionResult, evaluate_information_criteria, compute_pareto_front, select_optimal_model

struct ModelSelectionResult
    k::Int
    rss::Float64
    aic::Float64
    bic::Float64
    r2::Float64
    model::DiscoveredModel{Float64}
end

"""
    evaluate_information_criteria(model::DiscoveredModel, G::AbstractMatrix, b::AbstractVector)

Compute AIC, BIC, RSS, and R^2 for a DiscoveredModel given weak system matrices G and b.
"""
function evaluate_information_criteria(
    model::DiscoveredModel{Float64},
    G::AbstractMatrix{Float64},
    b::AbstractVector{Float64},
)
    N = length(b)
    k = model.sparsity_level

    # Reconstruct coefficient vector
    xi = zeros(Float64, size(G, 2))
    for term in model.terms
        # Map term back to column index if tracked, or compute residual directly
    end

    pred = G * xi
    res = b .- pred
    rss = dot(res, res)

    tss = dot(b .- sum(b)/N, b .- sum(b)/N)
    r2 = max(0.0, 1.0 - rss / max(tss, 1e-12))

    aic = 2 * k + N * log(max(rss / N, 1e-15))
    bic = k * log(N) + N * log(max(rss / N, 1e-15))

    return ModelSelectionResult(k, rss, aic, bic, r2, model)
end

"""
    compute_pareto_front(results::Vector{ModelSelectionResult})

Extract Pareto-optimal models minimizing model size k and RSS.
"""
function compute_pareto_front(results::Vector{ModelSelectionResult})
    sorted = sort(results; by = r -> (r.k, r.rss))
    pareto = ModelSelectionResult[]
    min_rss = Inf

    for res in sorted
        if res.rss < min_rss
            push!(pareto, res)
            min_rss = res.rss
        end
    end
    return pareto
end

end # module ModelSelection
