## Architectural Dependency DAG  
The strict unidirectional dependency hierarchy ensures Geometry operates as an autonomous mathematical engine independent of operator discovery:  
  
                      [ Observation ]  
                             │  
                             ▼  
                      [ Manifold ]  
                             │  
                             ▼  
                      [ Geometry ]  
                       /        \  
                      /          \  
                     ▼            ▼  
              [ Discovery ]   [ System (PrognosticPDE) ]  
                     \            /  
                      \          /  
                       ▼        ▼  
                      [ Closures ]  
                          │  
                          ▼  
                   [ Discretization ]  
* **Observation $\to$ Manifold:** Profile data is projected into state representation coordinates ($\eta_1, \eta_2, \eta_3, R, \Omega, \chi, \Pi_G, \lambda_{\min}$).   
* **Manifold $\to$ Geometry:** Geometry accepts intrinsic coordinates and algebraic vector fields to compute critical manifolds, fold curves, and Fenichel persistence without importing discovery code.   
* **Geometry $\to$ Discovery & System:** Discovery optionally queries geometric fold boundaries or tangent spaces for constraints, while System uses geometric Jacobians for stability metrics.   
* **Discovery & System $\to$ Closures:** Discovered coefficients or analytical models assemble into WSINDyClosureor MOSTClosure instances.   
* **Closures $\to$ Discretization:** The PDE system is discretized using MethodOfLinesFD or SpectralBLGalerkin.

**src/Geometry/Jacobians.jl**

Julia  
  
module Jacobians  
  
using Symbolics, LinearAlgebra, ModelingToolkit  
  
export JacobianCache, compute_fast_jacobian, compute_adjugate, evaluate_tangent_space  
  
struct JacobianCache  
    fast_vars::Vector{Num}  
    slow_vars::Vector{Num}  
    params::Vector{Num}  
    J_fast_symbolic::Matrix{Num}  
    det_J_symbolic::Num  
    adj_J_symbolic::Matrix{Num}  
end  
  
function JacobianCache(f_fast::Vector{Num}, fast_vars::Vector{Num}, slow_vars::Vector{Num}, params::Vector{Num})  
    J_fast = Symbolics.jacobian(f_fast, fast_vars)  
    det_J = Symbolics.det(J_fast)  
    # Symbolic 2x2 or 3x3 adjugate computation  
    adj_J = compute_symbolic_adjugate(J_fast)  
    return JacobianCache(fast_vars, slow_vars, params, J_fast, det_J, adj_J)  
end  
  
function compute_symbolic_adjugate(J::Matrix{Num})  
    n = size(J, 1)  
    if n == 2  
        return [J[2,2] -J[1,2]; -J[2,1] J[1,1]]  
    elseif n == 3  
        # Standard cofactor transpose matrix  
        cof = Matrix{Num}(undef, 3, 3)  
        cof[1,1] = J[2,2]*J[3,3] - J[2,3]*J[3,2]  
        cof[1,2] = -(J[2,1]*J[3,3] - J[2,3]*J[3,1])  
        cof[1,3] = J[2,1]*J[3,2] - J[2,2]*J[3,1]  
        cof[2,1] = -(J[1,2]*J[3,3] - J[1,3]*J[3,2])  
        cof[2,2] = J[1,1]*J[3,3] - J[1,3]*J[3,1]  
        cof[2,3] = -(J[1,1]*J[3,2] - J[1,2]*J[3,1])  
        cof[3,1] = J[1,2]*J[2,3] - J[1,3]*J[2,2]  
        cof[3,2] = -(J[1,1]*J[2,3] - J[1,3]*J[2,1])  
        cof[3,3] = J[1,1]*J[2,2] - J[1,2]*J[2,1]  
        return transpose(cof)  
    else  
        error("Adjugate for dimension $n > 3$ requires numeric factorization.")  
    end  
end  
  
end # module Jacobians

## Intermediate Representation: OperatorTerm  
Before emitting ModelingToolkit symbolic expressions, discovered terms are represented as structured metadata objects to enable simplification, dimensional verification, and cross-dataset comparison.  
  
Julia  
  
# Inside src/Discovery/SymbolicExtraction.jl  
  
export BasisOperator, OperatorTerm, DiscoveredModel, to_mtk_expression  
  
@enum OperatorType DIFFERENTIAL ALGEBRAIC DIAGNOSTIC  
  
struct BasisOperator  
    symbol::Symbol             # e.g., :u, :theta, :Ri, :Pi_G, :dz_u  
    type::OperatorType  
    spatial_derivative_order::Int  
    power::Float64  
end  
  
struct OperatorTerm  
    coefficient::Float64  
    basis::Vector{BasisOperator}  
end  
  
struct DiscoveredModel  
    target_variable::Symbol  
    terms::Vector{OperatorTerm}  
    residual_norm::Float64  
    sparsity_level::Int  
end  
  
function to_mtk_expression(model::DiscoveredModel, var_map::Dict{Symbol, Num})  
    expr = Num(0.0)  
    for term in model.terms  
        term_expr = Num(term.coefficient)  
        for b in term.basis  
            base_num = var_map[b.symbol]  
            if b.power != 1.0  
                term_expr *= base_num^b.power  
            else  
                term_expr *= base_num  
            end  
        end  
        expr += term_expr  
    end  
    return expr  
end

function evaluate_test_function(family::GegenbauerFamily, mode::Int, z::Float64)  
    # Maps z ∈ [z0, H] to [-1, 1] and evaluates C_n^(λ)(z) with compact boundary decay  
    z_ref = 2.0 * (z - z0) / (H - z0) - 1.0  
    return gegenbauer(mode, family.lambda, z_ref) * (1.0 - z_ref^2)^2  
end

**src/Discovery/SparseRegression.jl**  
Exposes an optimization interface supporting constrained thresholded ridge regression and quadratic programming.  
  
Julia  
  
module SparseRegression  
  
using JuMP, HiGHS, LinearAlgebra  
using ..ConstraintBuilder  
  
export AbstractSparseOptimizer, STRidge, ConstrainedQP, discover_coefficients  
  
abstract type AbstractSparseOptimizer end  
  
struct STRidge <: AbstractSparseOptimizer  
    lambda::Float64  
    threshold::Float64  
    max_iter::Int  
end  
  
struct ConstrainedQP <: AbstractSparseOptimizer  
    lambda::Float64  
    solver::Any  
end  
  
STRidge(; lambda=1e-3, threshold=1e-2, max_iter=50) = STRidge(lambda, threshold, max_iter)  
ConstrainedQP(; lambda=1e-3, solver=HiGHS.Optimizer) = ConstrainedQP(lambda, solver)  
  
function discover_coefficients(G::Matrix{Float64}, b::Vector{Float64},   
                               constraints::Union{Nothing, PhysicalConstraints},  
                               opt::STRidge)  
    # Standard STRidge loop: solve Ridge, threshold |ξ_i| < threshold, re-solve on support  
end  
  
function discover_coefficients(G::Matrix{Float64}, b::Vector{Float64},   
                               constraints::PhysicalConstraints,  
                               opt::ConstrainedQP)  
    model = Model(opt.solver)  
    set_silent(model)  
    n_features = size(G, 2)  
      
    @variable(model, xi[1:n_features])  
    @variable(model, abs_xi[1:n_features])  
      
    # Minimize 1/2 ||G*xi - b||^2 + lambda * sum(|xi|)  
    @objective(model, Min, 0.5 * sum((G * xi .- b).^2) + opt.lambda * sum(abs_xi))  
      
    @constraint(model, [i=1:n_features], xi[i] <= abs_xi[i])  
    @constraint(model, [i=1:n_features], -xi[i] <= abs_xi[i])  
      
    # Apply physical inequalities A_ineq * xi >= b_ineq  
    @constraint(model, constraints.A_ineq * xi .>= constraints.b_ineq)  
      
    optimize!(model)  
    return value.(xi)  
end  
  
end # module SparseRegression  
# 3. End-to-End Workflow Integration  
This example demonstrates data ingestion, weak-form assembly, constrained sparse optimization, conversion into a WSINDyClosure, and execution within the PDE solver.  
  
Julia  
  
using AtmosphericSlowManifold  
using AtmosphericSlowManifold.Observation  
using AtmosphericSlowManifold.Discovery  
using AtmosphericSlowManifold.Geometry  
using AtmosphericSlowManifold.System  
using AtmosphericSlowManifold.Discretization  
  
# 1. Ingest observation data  
obs = read_tower_csv("cabauw_profile_2026.csv")  
  
# 2. Build weak-form matrices  
test_basis = GegenbauerFamily(0.5, 10)  
library_features = [:u, :theta, :Ri, :Pi_G]  
weak_data = assemble_weak_matrices(obs, test_basis, library_features)  
  
# 3. Apply physical constraints (K_m >= 0)  
grid_samples = extract_evaluation_grid(obs)  
constraints = build_positivity_constraints(library_features, grid_samples)  
  
# 4. Perform sparse constrained regression  
coeffs = discover_coefficients(weak_data.G, weak_data.b, constraints, ConstrainedQP(lambda=1e-4))  
  
# 5. Extract intermediate representation and convert to MTK closure  
discovered_ir = DiscoveredModel(:K_m, coeffs, library_features)  
var_map = Dict(:u => u_sym, :theta => theta_sym, :Ri => Ri_sym, :Pi_G => Pi_G_sym)  
km_expr = to_mtk_expression(discovered_ir, var_map)  
  
discovered_closure = WSINDyClosure(km_expr, km_expr * 1.1, surface_flux_expr)  
  
# 6. Build prognostic PDE system and solve via finite difference or spectral backend  
pde_system = build_pde_system(discovered_closure)  
backend = MethodOfLinesFD(n_points=50, stretch_factor=2.5)  
solution = solve_scm(pde_system, discovered_closure, backend, (0.0, 86400.0))
