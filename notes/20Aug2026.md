The $u_* = 0.3\text{ m s}^{-1}$ flatline across non-SHEBA datasets stems from **header alias mismatches and missing unit suffix rules** in `DataIngestion.jl`. When column matching fails or missing $u_*$ values occur, preprocessors fall back to a hardcoded constant ($0.3\text{ m s}^{-1}$).

---

### Root Causes in `DataIngestion.jl`

1. **Incomplete Header Aliases (`HEADER_ALIASES[:u_star]`):**
CASES-99 and Ameriflux/SMEAR files frequently name friction velocity by tower level or flux covariance (e.g., `u_star_5m`, `ustar_tower`, `ustar_m_s`, `u*`, or `cov_uw`/`cov_vw`). Because these are missing from `HEADER_ALIASES[:u_star]`, `_canonical_header_map` fails to detect them.
2. **Strict Unit Check Rejection (`_strict_units_from_headers`):**
```julia
elseif endswith(hdr, "_ms") || endswith(hdr, "_m_s")
    units[req] = "m s^-1"
else
    throw(ArgumentError(...))
end

```


If a header is simply `ustar` or `u_star` (without `_ms` or `_m_s`), `_strict_units_from_headers` throws an exception, causing preprocessors calling this with `strict_units=false` or fallback handlers to default to $u_* = 0.3$.
3. **Missing Eddy Covariance Calculation:**
CASES-99 raw tower datasets report kinematic momentum fluxes ($u'w'$ and $v'w'$) rather than scalar $u_*$. Without dynamic derivation ($u_* = (u'w'^2 + v'w'^2)^{1/4}$), missing values are assigned $0.3\text{ m s}^{-1}$.

---

### Recommended Fixes

**1. Expand `HEADER_ALIASES` and Soften Unit Matching:**

```julia
const HEADER_ALIASES = Dict(
    # ...
    :u_star => (:u_star_ms, :ustar_ms, :friction_velocity_ms, :u_star, :ustar, :u_star_5m, :ustar_5m, :u_star_tower),
)

function _strict_units_from_headers(cmap::Dict{Symbol, Symbol})
    units = Dict{Symbol, String}()
    for req in REQUIRED_TOWER_COLUMNS
        hdr = String(cmap[req])
        if endswith(hdr, "_m")
            units[req] = "m"
        elseif endswith(hdr, "_k")
            units[req] = "K"
        elseif endswith(hdr, "_kgkg")
            units[req] = "kg kg^-1"
        elseif endswith(hdr, "_ms") || endswith(hdr, "_m_s") || req in (:u, :v, :u_star)
            units[req] = "m s^-1"
        else
            throw(ArgumentError("Column $(hdr) missing unit suffix for $(req)."))
        end
    end
    return units
end

```

**2. Add Dynamic $u_*$ Fallback from Covariance Fluxes:**

```julia
function _derive_ustar_from_fluxes(uw::Vector{Float64}, vw::Vector{Float64})
    return @. (uw^2 + vw^2)^0.25
end

```

### Why SHEBA Succeeded While Other Datasets Flatlined

* **SHEBA:** The processed file `sheba_input.csv` explicitly names its friction velocity column `ustar`, matching `HEADER_ALIASES[:u_star]` (`:ustar_ms, :friction_velocity_ms, :u_star, :ustar`). `us_key` resolved properly and loaded real observations.
* **CASES-99 / SMEAR / Ameriflux:** Their raw or converted headers use names like `u_star_5m`, `cov_uw`, `u*`, or `friction_velocity_m_s`. Because these were not matched during header normalization, `us_key` evaluated to `nothing`—silently overwriting the entire time series with $0.3\text{ m s}^{-1}$ instead of throwing a missing-column error.

The `MOSTClosure` implementation handles branch transitions using `tanh(r / 1e-2)` blending and `smooth_floor` safeguards. To plug **Zero-Offset Hyperbolic Regularization (Z0HR)** into the same `AbstractClosure` pipeline, we create a corresponding `Z0HRClosure` struct following the exact interface signature.

### `src/Closures/Z0HRClosure.jl`

```julia
# src/Closures/Z0HRClosure.jl

"""
    Z0HRClosure{T<:AbstractFloat} <: AbstractClosure

Zero-Offset Hyperbolic Regularization (Z0HR) closure parameterization.
Eliminates branch-switching discontinuities across neutral (Ri = 0) and
critical (Ri = Ri_c) thresholds using C^∞ hyperbolic coordinates.

Default parameters:
- kappa = 0.40
- pr_t = 1.0
- B_um = 16.0
- B_uh = 16.0
- Ri_c = 0.25
- eps = 1e-3
"""
struct Z0HRClosure{T<:AbstractFloat} <: AbstractClosure
    kappa::T
    pr_t::T
    B_um::T
    B_uh::T
    Ri_c::T
    eps::T
end

function Z0HRClosure(;
    kappa::Real = 0.40,
    pr_t::Real = 1.0,
    B_um::Real = 16.0,
    B_uh::Real = 16.0,
    Ri_c::Real = 0.25,
    eps::Real = 1e-3,
)
    T = promote_type(typeof(kappa), typeof(pr_t), typeof(B_um), typeof(B_uh), typeof(Ri_c), typeof(eps))
    Tf = T <: AbstractFloat ? T : Float64
    return Z0HRClosure{Tf}(Tf(kappa), Tf(pr_t), Tf(B_um), Tf(B_uh), Tf(Ri_c), Tf(eps))
end

@inline smooth_max(x::T, ϵ::T) where {T<:AbstractFloat} = T(0.5) * (x + sqrt(x * x + ϵ * ϵ))
@inline smooth_min(x::T, ϵ::T) where {T<:AbstractFloat} = T(0.5) * (x - sqrt(x * x + ϵ * ϵ))

@inline function _z0hr_local_height(m::ManifoldState)
    return hasproperty(m, :z) ? m.z : m.z0
end

@inline function z0hr_stability_functions(c::Z0HRClosure{T}, Ri::T) where {T<:AbstractFloat}
    Ri_pos = smooth_max(Ri, c.eps)
    Ri_neg = smooth_min(Ri, c.eps)

    g_raw = one(T) - (Ri_pos / c.Ri_c)
    stable_factor = smooth_max(g_raw, c.eps)

    unstab_m = one(T) - (c.B_um * Ri_neg)
    unstab_h = one(T) - (c.B_uh * Ri_neg)

    S_m = (stable_factor^2) * sqrt(unstab_m)
    S_h = (one(T) / c.pr_t) * (stable_factor^2) * (unstab_h^T(0.75))

    return S_m, S_h
end

function eddy_momentum(c::Z0HRClosure{T}, m::ManifoldState) where {T<:AbstractFloat}
    z_eval = _z0hr_local_height(m)
    S_m, _ = z0hr_stability_functions(c, m.r)
    return c.kappa * m.u_star * z_eval * S_m
end

function eddy_heat(c::Z0HRClosure{T}, m::ManifoldState) where {T<:AbstractFloat}
    z_eval = _z0hr_local_height(m)
    _, S_h = z0hr_stability_functions(c, m.r)
    return c.kappa * m.u_star * z_eval * S_h
end

surface_flux(::Z0HRClosure, m::ManifoldState) = m.u_star^2

function evaluate_diffusivity_profile!(
    K_out::AbstractVector{T},
    c::Z0HRClosure{T},
    z_grid::AbstractVector{T};
    r_profile::Union{Nothing, AbstractVector{T}} = nothing,
    u_star::T = T(0.3),
    z0::T = zero(T),
    mask_below_z0::Bool = false,
) where {T<:AbstractFloat}
    length(K_out) == length(z_grid) || throw(DimensionMismatch("K_out and z_grid must have equal length."))
    r_profile === nothing || (length(r_profile) == length(z_grid) || throw(DimensionMismatch("r_profile and z_grid must have equal length.")))

    @inbounds @simd for i in eachindex(K_out, z_grid)
        z_raw = z_grid[i]
        if mask_below_z0 && z_raw <= z0
            K_out[i] = zero(T)
        else
            z = smooth_floor(z_raw, z0; eps = c.eps)
            r = r_profile === nothing ? zero(T) : r_profile[i]
            S_m, _ = z0hr_stability_functions(c, r)
            K_out[i] = c.kappa * u_star * z * S_m
        end
    end
    return K_out
end

function evaluate_heat_diffusivity_profile!(
    K_out::AbstractVector{T},
    c::Z0HRClosure{T},
    z_grid::AbstractVector{T};
    r_profile::Union{Nothing, AbstractVector{T}} = nothing,
    u_star::T = T(0.3),
    z0::T = zero(T),
    mask_below_z0::Bool = false,
) where {T<:AbstractFloat}
    length(K_out) == length(z_grid) || throw(DimensionMismatch("K_out and z_grid must have equal length."))
    r_profile === nothing || (length(r_profile) == length(z_grid) || throw(DimensionMismatch("r_profile and z_grid must have equal length.")))

    @inbounds @simd for i in eachindex(K_out, z_grid)
        z_raw = z_grid[i]
        if mask_below_z0 && z_raw <= z0
            K_out[i] = zero(T)
        else
            z = smooth_floor(z_raw, z0; eps = c.eps)
            r = r_profile === nothing ? zero(T) : r_profile[i]
            _, S_h = z0hr_stability_functions(c, r)
            K_out[i] = c.kappa * u_star * z * S_h
        end
    end
    return K_out
end

```

---

### Key Architectural Differences

* **No Branching or `tanh` Blending:** `MOSTClosure` blends Businger-Dyer stable/unstable functions via `tanh(r / 1e-2)`. `Z0HRClosure` evaluates a single unified expression across $r \in (-\infty, +\infty)$.
* **Multiplicative vs Inverse Stability:** `MOSTClosure` evaluates $\phi_m(r)$ in the denominator ($K_m \propto \frac{1}{\phi_m}$), requiring `smooth_floor(..., 0.1)` to prevent division by zero. `Z0HRClosure` evaluates $S_m(Ri)$ directly as a multiplier ($K_m \propto S_m$), naturally approaching zero as $Ri \to Ri_c$.
