### Root Cause & Fix Location

#### Fix in `src/Benchmark/PDEBenchmark.jl`

Ensure `evaluate_candidate` computes profile metrics against the observational baseline matrix $U_{\text{obs}}(z, t)$:

```julia
function evaluate_candidate(candidate_name::String, model::DiscoveredModel, prob::ODEProblem, u_obs::Matrix{Float64}, t_grid::Vector{Float64})
    sol = solve(prob, Rodas5P(autodiff=true); reltol=1e-6, abstol=1e-8, maxiters=1_000_000)

    if sol.retcode != :Success && sol.retcode != :Terminated
        @warn "Simulation failed for candidate $candidate_name with return code $(sol.retcode)"
        return nothing
    end

    # Extract predicted u(z, t) matrix from ODE solution
    u_pred = Array(sol) # Size: (n_z, n_t)

    # Compute residual metrics
    res = u_obs .- u_pred
    rss = sum(res .^ 2)
    res_norm = sqrt(rss)

    u_mean = mean(u_obs)
    tss = sum((u_obs .- u_mean) .^ 2)
    r2 = 1.0 - (rss / max(tss, 1e-12))

    n_obs = length(u_obs)
    k = model.sparsity
    aic = n_obs * log(max(rss / n_obs, 1e-12)) + 2k
    bic = n_obs * log(max(rss / n_obs, 1e-12)) + k * log(n_obs)

    return (
        candidate = candidate_name,
        k = k,
        rss = rss,
        r2 = r2,
        residual_norm = res_norm,
        aic = aic,
        bic = bic,
        pareto_rss = false,
        pareto_r2 = false
    )
end

```
