### 1. Visual Overlap Plotting Fix

Using a combination of alpha blending, offset line widths, and distinct markers prevents solid lines from masking dashed baseline trajectories:

```julia
using Plots
using Plots.PlotMeasures

p = plot(
    layout = (1, 2),
    size = (1200, 520),
    bottom_margin = 12mm,
    left_margin = 8mm,
)

for (j, campaign) in enumerate(sort(collect(keys(profiles))))
    prof = profiles[campaign]
    mode_idx = collect(1:length(prof.observed))

    # Observed Profile
    plot!(
        p[j],
        mode_idx,
        prof.observed,
        lw = 2,
        color = :black,
        label = "Observed modal profile",
        xlabel = "Mode Index",
        ylabel = "Amplitude",
        title = "$(campaign) Modal Profile",
    )

    # Neutral Baseline (Thicker dashed line with alpha to show behind physical)
    plot!(
        p[j],
        mode_idx,
        prof.final_base,
        lw = 3.5,
        ls = :dash,
        color = :red,
        alpha = 0.6,
        label = "Neutral baseline",
    )

    # Physical Closure (Solid thin line with distinct markers)
    plot!(
        p[j],
        mode_idx,
        prof.final_phys,
        lw = 1.5,
        ls = :solid,
        color = :blue,
        marker = :circle,
        ms = 3,
        label = "Physical closure",
    )
end

```
