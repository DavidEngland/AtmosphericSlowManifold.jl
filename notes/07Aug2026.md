`AtmosphericSlowManifold.jl` (ASM.jl) provides a data-driven, mathematically rigorous alternative to classical atmospheric boundary layer (ABL) turbulence closures by unifying Geometric Singular Perturbation Theory (GSPT) with Weak-form Sparse Identification of Non-linear Dynamics (WSINDy).

using Symbolics

export to_latex, latex_term_table, latex_site_summary_table

const _GREEK_MAP = Dict(
    "alpha" => "\\alpha",
    "beta" => "\\beta",
    "gamma" => "\\gamma",
    "delta" => "\\delta",
    "epsilon" => "\\epsilon",
    "theta" => "\\theta",
    "kappa" => "\\kappa",
    "lambda" => "\\lambda",
    "mu" => "\\mu",
    "nu" => "\\nu",
    "xi" => "\\xi",
    "pi" => "\\pi",
    "rho" => "\\rho",
    "sigma" => "\\sigma",
    "tau" => "\\tau",
    "phi" => "\\phi",
    "psi" => "\\psi",
    "omega" => "\\omega",
)

function _num_to_latex(x::Real)
    if iszero(x)
        return "0"
    end

    if isinteger(x)
        return string(Int(round(x)))
    end

    xs = string(round(Float64(x); sigdigits = 6))
    if occursin("e", xs)
        parts = split(xs, "e")
        base = parts[1]
        expo = parse(Int, parts[2])
        return string(base, "\\times 10^{", expo, "}")
    end
    return xs
end

function _latex_symbol(sym::Symbol)
    s = String(sym)
    if haskey(_GREEK_MAP, lowercase(s))
        return _GREEK_MAP[lowercase(s)]
    elseif length(s) == 1
        return s
    end

    escaped = replace(s, "_" => "\\_")
    return string("\\mathrm{", escaped, "}")
end

function _feature_to_latex(f::StateVariable)
    return _latex_symbol(f.name)
end

function _feature_to_latex(f::DiagnosticVariable)
    return _latex_symbol(f.name)
end

function _feature_to_latex(f::SpatialDerivative)
    var = _latex_symbol(f.variable)
    z = "z"
    if f.order <= 1
        return string("\\frac{\\partial ", var, "}{\\partial ", z, "}")
    end
    return string("\\frac{\\partial^{", f.order, "} ", var, "}{\\partial ", z, "^{", f.order, "}}")
end

function _basis_to_latex(b::BasisOperator)
    base = _feature_to_latex(b.feature)
    p = b.power
    if abs(p - 1.0) <= 1e-12
        return base
    end

    ptxt = isinteger(p) ? string(Int(round(p))) : _num_to_latex(p)
    return string("\\left(", base, "\\right)^{", ptxt, "}")
end

function _term_magnitude_to_latex(term::OperatorTerm)
    c = abs(Float64(term.coefficient))
    factors = [_basis_to_latex(b) for b in term.basis]

    if isempty(factors)
        return _num_to_latex(c)
    end

    if abs(c - 1.0) <= 1e-12
        return join(factors, " ")
    end

    return string(_num_to_latex(c), " ", join(factors, " "))
end

function _rhs_to_latex(model::DiscoveredModel)
    if isempty(model.terms)
        return "0"
    end

    chunks = String[]
    for (i, term) in enumerate(model.terms)
        sign = Float64(term.coefficient) < 0 ? "-" : "+"
        mag = _term_magnitude_to_latex(term)

        if i == 1
            if sign == "-"
                push!(chunks, string("-", mag))
            else
                push!(chunks, mag)
            end
        else
            push!(chunks, string(" ", sign, " ", mag))
        end
    end
    return join(chunks)
end

"""
    to_latex(model::DiscoveredModel; independent=:t)

Create publication-ready LaTeX for discovered prognostic equations.
"""
function to_latex(model::DiscoveredModel; independent::Symbol = :t)
    lhs = string(
        "\\frac{\\partial ",
        _latex_symbol(model.target_variable),
        "}{\\partial ",
        _latex_symbol(independent),
        "}",
    )
    rhs = _rhs_to_latex(model)
    return string(lhs, " = ", rhs)
end

function _symbolics_text_to_latex(expr)
    s = string(expr)
    s = replace(s, "θ" => "\\theta")
    s = replace(s, "ν" => "\\nu")
    s = replace(s, "ξ" => "\\xi")
    s = replace(s, "κ" => "\\kappa")
    return replace(s, "_" => "\\_")
end

"""
    to_latex(closure::WSINDyClosure; field=:momentum)

Return LaTeX for one closure expression (`:momentum`, `:heat`, or `:flux`).
"""
function to_latex(closure::WSINDyClosure; field::Symbol = :momentum)
    expr = if field == :momentum
        closure.km_expr
    elseif field == :heat
        closure.kh_expr
    elseif field == :flux
        closure.flux_expr
    else
        throw(ArgumentError("Unknown WSINDyClosure field $(field). Use :momentum, :heat, or :flux."))
    end
    return _symbolics_text_to_latex(expr)
end

"""
    latex_term_table(model::DiscoveredModel; r2=nothing, residual_norm=model.residual_norm)

Generate a LaTeX tabular environment for discovered terms and coefficients.
"""
function latex_term_table(
    model::DiscoveredModel;
    r2::Union{Nothing, Real} = nothing,
    residual_norm::Real = model.residual_norm,
)
    lines = String[]
    push!(lines, "\\begin{tabular}{r l r}")
    push!(lines, "\\hline")
    push!(lines, "Term & Basis & Coefficient \\\\")
    push!(lines, "\\hline")

    for (i, term) in enumerate(model.terms)
        basis = isempty(term.basis) ? "1" : join([_basis_to_latex(b) for b in term.basis], " ")
        coeff = _num_to_latex(Float64(term.coefficient))
        push!(lines, string(i, " & \$", basis, "\$ & \$", coeff, "\$ \\\\"))
    end

    push!(lines, "\\hline")
    if r2 !== nothing
        push!(lines, string("\\multicolumn{3}{l}{\$R^{2}=", _num_to_latex(Float64(r2)), "\$} \\\\"))
    end
    push!(lines, string("\\multicolumn{3}{l}{Residual norm=", _num_to_latex(Float64(residual_norm)), "} \\\\"))
    push!(lines, "\\hline")
    push!(lines, "\\end{tabular}")
    return join(lines, "\n")
end

"""
    latex_site_summary_table(models; r2_by_site=Dict(), residual_by_site=Dict())

Create a publication table comparing fit quality across sites.
"""
function latex_site_summary_table(
    models::AbstractDict{Symbol, <:DiscoveredModel};
    r2_by_site::Dict{Symbol, <:Real} = Dict{Symbol, Float64}(),
    residual_by_site::Dict{Symbol, <:Real} = Dict{Symbol, Float64}(),
)
    lines = String[]
    push!(lines, "\\begin{tabular}{l r r r}")
    push!(lines, "\\hline")
    push!(lines, "Site & Active terms & \$R^{2}\$ & Residual norm \\\\")
    push!(lines, "\\hline")

    for site in sort(collect(keys(models)); by = String)
        m = models[site]
        n_active = m.sparsity_level
        r2 = haskey(r2_by_site, site) ? _num_to_latex(Float64(r2_by_site[site])) : "-"
        rn = haskey(residual_by_site, site) ? _num_to_latex(Float64(residual_by_site[site])) : _num_to_latex(m.residual_norm)
        push!(lines, string(_latex_symbol(site), " & ", n_active, " & ", r2, " & ", rn, " \\\\"))
    end

    push!(lines, "\\hline")
    push!(lines, "\\end{tabular}")
    return join(lines, "\n")
end

"""
Format floating point or integer values for LaTeX display using robust precision control.
"""
function _num_to_latex(x::Real; sigdigits::Int = 6)
    if iszero(x)
        return "0"
    end
    if isinteger(x)
        return string(Int(round(x)))
    end

    formatted = @sprintf("%.*g", sigdigits, Float64(x))
    if occursin("e", formatted)
        parts = split(formatted, "e")
        base = parts[1]
        expo = parse(Int, parts[2])
        return string(base, "\\times 10^{", expo, "}")
    end
    return formatted
end

"""
Parse symbols with support for Greek characters, multi-character subscripts, and special characters.
Examples: :theta_v -> \\theta_{v}, :Ri_g -> \\mathrm{Ri}_{g}, :SHEBA_2001 -> \\mathrm{SHEBA}_{2001}
"""
function _latex_symbol(sym::Symbol)
    s = String(sym)
    parts = split(s, "_")
    head = parts[1]

    formatted_head = if haskey(_GREEK_MAP, head)
        _GREEK_MAP[head]
    elseif haskey(_GREEK_MAP, lowercase(head))
        _GREEK_MAP[lowercase(head)]
    elseif length(head) == 1
        head
    else
        string("\\mathrm{", replace(head, "-" => "\\text{-}"), "}")
    end

    if length(parts) == 1
        return formatted_head
    end

    # Process subscript components
    sub_raw = join(parts[2:end], "\\_")
    formatted_sub = if haskey(_GREEK_MAP, sub_raw)
        _GREEK_MAP[sub_raw]
    elseif length(sub_raw) == 1
        sub_raw
    else
        string("\\mathrm{", replace(sub_raw, "-" => "\\text{-}"), "}")
    end

    return string(formatted_head, "_{", formatted_sub, "}")
end

    var = _latex_symbol(f.variable)
    # Dynamically extract spatial coordinate variable, default to :z
    coord_sym = hasproperty(f, :coordinate) ? f.coordinate : :z
    coord = _latex_symbol(coord_sym)

    if f.order <= 1
        return string("\\frac{\\partial ", var, "}{\\partial ", coord, "}")
    end
    return string("\\frac{\\partial^{", f.order, "} ", var, "}{\\partial ", coord, "^{", f.order, "}}")
end

function _basis_to_latex(b::BasisOperator)
    base = _feature_to_latex(b.feature)
    p = b.power
    if abs(p - 1.0) <= 1e-12
        return base
    end

    ptxt = isinteger(p) ? string(Int(round(p))) : _num_to_latex(p)
    return string("\\left(", base, "\\right)^{", ptxt, "}")
end

function _term_magnitude_to_latex(term::OperatorTerm)
    c = abs(Float64(term.coefficient))
    factors = [_basis_to_latex(b) for b in term.basis]

    if isempty(factors)
        return _num_to_latex(c)
    end

    # Join factors with thin space \\, for standard journal typesetting
    factors_str = join(factors, "\\,")

    if abs(c - 1.0) <= 1e-12
        return factors_str
    end

    return string(_num_to_latex(c), "\\,", factors_str)
end

function _rhs_to_latex(model::DiscoveredModel)
    if isempty(model.terms)
        return "0"
    end

    chunks = String[]
    for (i, term) in enumerate(model.terms)
        sign = Float64(term.coefficient) < 0 ? "-" : "+"
        mag = _term_magnitude_to_latex(term)

        if i == 1
            if sign == "-"
                push!(chunks, string("-", mag))
            else
                push!(chunks, mag)
            end
        else
            push!(chunks, string(" ", sign, " ", mag))
        end
    end
    return join(chunks)
end

function _wrap_environment(eq::String, env::Symbol)
    if env == :none
        return eq
    elseif env in (:equation, :eq)
        return string("\\begin{equation}\n", eq, "\n\\end{equation}")
    elseif env in (:equation_star, :eq_star, Symbol("equation*"))
        return string("\\begin{equation*}\n", eq, "\n\\end{equation*}")
    elseif env == :align
        return string("\\begin{align}\n", eq, "\n\\end{align}")
    elseif env in (:align_star, Symbol("align*"))
        return string("\\begin{align*}\n", eq, "\n\\end{align*}")
    else
        throw(ArgumentError("Unknown environment $(env). Supported: :none, :equation, :equation_star, :align, :align_star."))
    end
end

"""
    to_latex(model::DiscoveredModel; independent=:t, environment=:none)

Create publication-ready LaTeX for discovered prognostic equations.
Supported `environment` values: `:none`, `:equation`, `:equation_star`, `:align`, `:align_star`.
"""
function to_latex(
    model::DiscoveredModel;
    independent::Symbol = :t,
    environment::Symbol = :none
)
    lhs = string(
        "\\frac{\\partial ",
        _latex_symbol(model.target_variable),
        "}{\\partial ",
        _latex_symbol(independent),
        "}",
    )
    rhs = _rhs_to_latex(model)
    eq_str = string(lhs, " = ", rhs)

    return _wrap_environment(eq_str, environment)
end

"""
    to_latex(models::AbstractVector{<:DiscoveredModel}; independent=:t, environment=:align)

Export a multi-equation system of discovered models (e.g., u, θ, q) formatted for LaTeX alignment environments.
"""
function to_latex(
    models::AbstractVector{<:DiscoveredModel};
    independent::Symbol = :t,
    environment::Symbol = :align
)
    lines = String[]
    for m in models
        lhs = string(
            "\\frac{\\partial ",
            _latex_symbol(m.target_variable),
            "}{\\partial ",
            _latex_symbol(independent),
            "}"
        )
        rhs = _rhs_to_latex(m)
        push!(lines, string(lhs, " &= ", rhs))
    end
    body = join(lines, " \\\\\n")

    if environment == :align
        return string("\\begin{align}\n", body, "\n\\end{align}")
    elseif environment in (:align_star, Symbol("align*"))
        return string("\\begin{align*}\n", body, "\n\\end{align*}")
    else
        return body
    end
end

"""
    to_latex(closure::WSINDyClosure; field=:momentum)

Return LaTeX for symbolic closure expressions using `Latexify.jl`.
"""
function to_latex(closure::WSINDyClosure; field::Symbol = :momentum)
    expr = if field == :momentum
        closure.km_expr
    elseif field == :heat
        closure.kh_expr
    elseif field == :flux
        closure.flux_expr
    else
        throw(ArgumentError("Unknown WSINDyClosure field $(field). Use :momentum, :heat, or :flux."))
    end

    return String(latexify(expr))
end

"""
    latex_term_table(
        model::DiscoveredModel;
        r2=nothing,
        residual_norm=model.residual_norm,
        booktabs=true,
        longtable=false
    )

Generate a LaTeX `tabular` or `longtable` environment for discovered terms and coefficients.
"""
function latex_term_table(
    model::DiscoveredModel;
    r2::Union{Nothing, Real} = nothing,
    residual_norm::Real = model.residual_norm,
    booktabs::Bool = true,
    longtable::Bool = false
)
    table_env = longtable ? "longtable" : "tabular"
    top_rule = booktabs ? "\\toprule" : "\\hline"
    mid_rule = booktabs ? "\\midrule" : "\\hline"
    bot_rule = booktabs ? "\\bottomrule" : "\\hline"

    lines = String[]
    push!(lines, string("\\begin{", table_env, "}{r l r}"))
    push!(lines, top_rule)
    push!(lines, "Term & Basis & Coefficient \\\\")
    push!(lines, mid_rule)

    for (i, term) in enumerate(model.terms)
        basis = isempty(term.basis) ? "1" : join([_basis_to_latex(b) for b in term.basis], "\\,")
        coeff = _num_to_latex(Float64(term.coefficient))
        push!(lines, string(i, " & \$", basis, "\$ & \$", coeff, "\$ \\\\"))
    end

    push!(lines, mid_rule)
    if r2 !== nothing
        push!(lines, string("\\multicolumn{3}{l}{\$R^{2} = ", _num_to_latex(Float64(r2)), "\$} \\\\"))
    end
    push!(lines, string("\\multicolumn{3}{l}{Residual norm = ", _num_to_latex(Float64(residual_norm)), "} \\\\"))
    push!(lines, bot_rule)
    push!(lines, string("\\end{", table_env, "}"))
    return join(lines, "\n")
end

```julia
function compute_km_profile_uncertainty(df::DataFrame, z_grid::Vector{Float64}; n_draws::Int = 1000)
    Random.seed!(42)
    ustar_mean = hasproperty(df, :ustar) ? finite_mean(get_numeric_column(df, :ustar)) : 0.25
    ustar_mean = isfinite(ustar_mean) && ustar_mean > 0 ? ustar_mean : 0.25

    L_mean = hasproperty(df, :L_obukhov) ? finite_mean(get_numeric_column(df, :L_obukhov)) : 50.0
    L_mean = isfinite(L_mean) && abs(L_mean) > 1.0 ? L_mean : 50.0

    kappa_draws = randn(n_draws) .* 0.02 .+ 0.40
    beta_draws = randn(n_draws) .* 0.50 .+ 4.70
    ustar_draws = max.(0.01, randn(n_draws) .* 0.05 .+ ustar_mean)

    n_z = length(z_grid)
    km_draws = zeros(Float64, n_z, n_draws)

    for d in 1:n_draws
        κ = kappa_draws[d]
        β = beta_draws[d]
        u_s = ustar_draws[d]
        for (i, z) in enumerate(z_grid)
            ζ = z / L_mean
            ϕ_m = ζ >= 0 ? (1.0 + β * ζ) : (1.0 - 16.0 * ζ)^(-0.25)
            km_draws[i, d] = (κ * u_s * z) / max(0.1, ϕ_m)
        end
    end

    q25 = [quantile(km_draws[i, :], 0.025) for i in 1:n_z]
    q50 = [quantile(km_draws[i, :], 0.500) for i in 1:n_z]
    q975 = [quantile(km_draws[i, :], 0.975) for i in 1:n_z]

    return q25, q50, q975
end
```

## 1. Geometric Singular Perturbation (GSPT) Metrics Integration

### Diagnostic Computation (`src/diagnostics/gspt.jl`)

```julia
module GSPTDiagnostics

using LinearAlgebra, ForwardDiff

export compute_transversality, compute_fold_distance

"""
    compute_transversality(u, v, grad_h_fn)

Calculates the transversality indicator T(z,t) = ∇h(u) ⋅ v, measuring vector field
alignment relative to the normal hyperbolicity manifold boundary h(u) = 0.
"""
function compute_transversality(u_state::AbstractVector{T}, v_drift::AbstractVector{T}, grad_h_fn::Function) where T<:Real
    grad_h = grad_h_fn(u_state)
    return dot(grad_h, v_drift)
end

"""
    compute_fold_distance(u_state, fold_manifold)

Evaluates minimum Euclidean distance in state space d_fold(z,t) = || x(z,t) - x_fold ||_2
between current flow state and critical fold set.
"""
function compute_fold_distance(u_state::AbstractVector{T}, fold_manifold::AbstractMatrix{T}) where T<:Real
    min_dist = typemax(T)
```

```julia
function evaluate_candidate_model(rss::Real, k::Int, N::Int)
    rss_safe = max(rss, eps(Float64))
    aic = 2 * k + N * log(rss_safe / N)
    bic = k * log(N) + N * log(rss_safe / N)
    return (aic = aic, bic = bic)
end

function kfold_cross_validate(model_fit_fn::Function, predict_fn::Function, X::Array{T,2}, y::Vector{T}; k_folds::Int=5) where T<:Real
    N = length(y)
    fold_sizes = fill(div(N, k_folds), k_folds)
    fold_sizes[1:mod(N, k_folds)] .+= 1

    indices = randperm(N)
    starts = [1; cumsum(fold_sizes)[1:end-1] .+ 1]
    stops = cumsum(fold_sizes)

    rss_cv = 0.0
    for fold in 1:k_folds
        test_idx = indices[starts[fold]:stops[fold]]
        train_idx = setdiff(indices, test_idx)

        fitted_model = model_fit_fn(X[train_idx, :], y[train_idx])
        y_pred = predict_fn(fitted_model, X[test_idx, :])
        rss_cv += sum((y[test_idx] .- y_pred).^2)
    end
    rmse_cv = sqrt(rss_cv / N)
    return (rss_cv = rss_cv, rmse_cv = rmse_cv)
end

end
```

```julia
    F_bc::Vector{Float64},
    U_1::Float64,
    Theta_1::Float64,
    Theta_s::Float64,
    params::NamedTuple
)
    rho = params.rho
    c_p = params.c_p
    C_D = params.C_D
    C_H = params.C_H

    # Surface momentum stress and sensible heat flux
    tau_s = -rho * C_D * U_1^2
    q_s   = -rho * c_p * C_H * U_1 * (Theta_1 - Theta_s)
```

```julia
module RegularityDiagnostic

using ForwardDiff, LinearAlgebra, Statistics, Test

export verify_regularity_and_solver_performance

"""
    verify_regularity_and_solver_performance(closure_fn, residual_fn, u0, domain; max_derivative=4)

Evaluates order k ∈ [1, 4] derivatives, estimates local Lipschitz constants, computes
Jacobian condition numbers κ(J), and measures Newton-Raphson iteration counts.
"""
function verify_regularity_and_solver_performance(
    closure_fn::Function,
    residual_fn!::Function,
    u0::Vector{T},
    domain::AbstractVector{T};
    max_derivative::Int=4,
    tol::T=1e-8,
    max_iters::Int=50
) where T<:Real

    # 1. High-Order Algorithmic Differentiation
    ders = Dict{Int, Vector{T}}()
    ders[0] = [closure_fn(x) for x in domain]
    current_fn = closure_fn

    for k in 1:max_derivative
        df(x) = ForwardDiff.derivative(current_fn, x)
        ders[k] = [df(x) for x in domain]
        current_fn = df
    end

    # 2. Lipschitz Constant Estimation: L = max |dK/dζ|
    lipschitz_constant = maximum(abs.(ders[1]))
    max_deriv_magnitudes = [maximum(abs.(ders[k])) for k in 1:max_derivative]

    # 3. Jacobian Condition Number κ(J) across Domain
    cond_numbers = Vector{T}(undef, length(domain))
    J_buffer = zeros(T, length(u0), length(u0))
    for (i, x) in enumerate(domain)
        # Compute Jacobian of PDE residual at state x
        res_at_x(u) = (out = similar(u); residual_fn!(out, u, x); out)
        J = ForwardDiff.jacobian(res_at_x, u0)
        cond_numbers[i] = cond(J)
    end
    max_cond = maximum(filter(x -> !isnan(x) && !isinf(x), cond_numbers))

    # 4. Newton-Raphson Iteration Metrics
    iters_required = Int[]
    failures = 0
    for x in domain
        u = copy(u0)
        res = zeros(T, length(u0))
        converged = false

        for iter in 1:max_iters
            residual_fn!(res, u, x)
            if norm(res, 2) < tol
                push!(iters_required, iter)
                converged = true
                break
            end
            J = ForwardDiff.jacobian(u_vec -> (out = similar(u_vec); residual_fn!(out, u_vec, x); out), u)
            delta = J \ (-res)
            u .+= delta
        end
        if !converged
            failures += 1
            push!(iters_required, max_iters)
        end
    end

    return (
        derivatives = ders,
        max_derivative_magnitudes = max_deriv_magnitudes,
        lipschitz_constant = lipschitz_constant,
        max_jacobian_cond = max_cond,
        mean_newton_iters = mean(iters_required),
        max_newton_iters = maximum(iters_required),
        solver_failure_rate = failures / length(domain)
    )
end

end
```

```julia
struct ModalBudget{T<:Real}

    F_adv::Vector{T}
    F_diff::Vector{T}
    F_lin::Vector{T}
    F_bc::Vector{T}
    residual::Vector{T}
end

"""
Decomposes Gegenbauer expansion modal tendencies:
da_n/dt = F_adv^(n) + F_diff^(n) + F_lin^(n) + F_bc^(n)
"""
function compute_modal_tendencies!(
    budget::ModalBudget{T},
    a_n::Vector{T},
    F_adv::Vector{T},
    F_diff::Vector{T},
    F_lin::Vector{T},
    tau_s::T,
    q_s::T,
    basis_weights::Vector{T}
) where T<:Real

    num_modes = length(a_n)

    # Project surface stress τ_s and sensible heat flux q_s onto modal basis
    @inbounds for n in 1:num_modes
        budget.F_bc[n] = (tau_s + q_s) * basis_weights[n]
        budget.F_adv[n] = F_adv[n]
        budget.F_diff[n] = F_diff[n]
        budget.F_lin[n] = F_lin[n]

        # Balance equation
        budget.da_dt[n] = budget.F_adv[n] + budget.F_diff[n] + budget.F_lin[n] + budget.F_bc[n]
        budget.residual[n] = budget.da_dt[n] - (budget.F_adv[n] + budget.F_diff[n] + budget.F_lin[n] + budget.F_bc[n])
    end

    @assert maximum(abs.(budget.residual)) < 1e-10 "Conservation budget violation!"
    return budget
end
```

```julia
module StatisticalValidation

using LinearAlgebra, Statistics, Random

export ModelValidationMetrics, evaluate_full_statistical_suite

struct ModelValidationMetrics{T<:Real}
    rss::T
    r2::T
    adj_r2::T
    aic::T
    bic::T
    press::T
    rmse_cv::T
    param_ci_lower::Vector{T}
    param_ci_upper::Vector{T}
end

"""
    evaluate_full_statistical_suite(X, y, θ_hat, fit_fn, predict_fn; k_folds=5, n_bootstraps=200)

Calculates Adjusted R², AIC, BIC, PRESS, K-Fold RMSE, and 95% Bootstrap Confidence Intervals.
"""
function evaluate_full_statistical_suite(
    X::Matrix{T},
    y::Vector{T},
    θ_hat::Vector{T},
    fit_fn::Function,
    predict_fn::Function;
    k_folds::Int=5,
    n_bootstraps::Int=200
) where T<:Real

    N, k = size(X)
    y_pred = predict_fn(θ_hat, X)
    residuals = y .- y_pred
    rss = sum(residuals .^ 2)
    tss = sum((y .- mean(y)) .^ 2)

    # Standard R² and Adjusted R²
    r2 = 1.0 - (rss / tss)
    adj_r2 = 1.0 - ((1.0 - r2) * (N - 1) / (N - k - 1))

    # Information Criteria
    rss_safe = max(rss, eps(T))
    aic = 2 * k + N * log(rss_safe / N)
    bic = k * log(N) + N * log(rss_safe / N)

    # PRESS Statistic (Prediction Residual Sum of Squares) via Hat Matrix
    H = X * ((X' * X) \ X')
    press_residuals = residuals ./ (1.0 .- diag(H))
    press = sum(press_residuals .^ 2)

    # K-Fold Cross-Validation Residual Score
    fold_sizes = fill(div(N, k_folds), k_folds)
    fold_sizes[1:mod(N, k_folds)] .+= 1
    indices = randperm(N)
    starts = [1; cumsum(fold_sizes)[1:end-1] .+ 1]
    stops = cumsum(fold_sizes)

    rss_cv = 0.0
    for fold in 1:k_folds
        test_idx = indices[starts[fold]:stops[fold]]
        train_idx = setdiff(indices, test_idx)

        θ_fold = fit_fn(X[train_idx, :], y[train_idx])
        y_cv_pred = predict_fn(θ_fold, X[test_idx, :])
        rss_cv += sum((y[test_idx] .- y_cv_pred) .^ 2)
    end
    rmse_cv = sqrt(rss_cv / N)

    # Bootstrap 95% Confidence Intervals
    boot_params = zeros(T, n_bootstraps, k)
    for b in 1:n_bootstraps
        boot_idx = rand(1:N, N)
        boot_params[b, :] = fit_fn(X[boot_idx, :], y[boot_idx])
    end

    ci_lower = [quantile(boot_params[:, j], 0.025) for j in 1:k]
    ci_upper = [quantile(boot_params[:, j], 0.975) for j in 1:k]

    return ModelValidationMetrics(
        rss, r2, adj_r2, aic, bic, press, rmse_cv, ci_lower, ci_upper
    )
end

end
```

```julia
module RegularityDiagnostic

using ForwardDiff, LinearAlgebra, Statistics, Test

export verify_regularity_and_solver_performance

"""
    verify_regularity_and_solver_performance(closure_fn, residual_fn, u0, domain; max_derivative=4)

Evaluates order k ∈ [1, 4] derivatives, estimates local Lipschitz constants, computes
Jacobian condition numbers κ(J), and measures Newton-Raphson iteration counts.
"""
function verify_regularity_and_solver_performance(
    closure_fn::Function,
    residual_fn!::Function,
    u0::Vector{T},
    domain::AbstractVector{T};
    max_derivative::Int=4,
    tol::T=1e-8,
    max_iters::Int=50
) where T<:Real

    # 1. High-Order Algorithmic Differentiation
    ders = Dict{Int, Vector{T}}()
    ders[0] = [closure_fn(x) for x in domain]
    current_fn = closure_fn

    for k in 1:max_derivative
        df(x) = ForwardDiff.derivative(current_fn, x)
        ders[k] = [df(x) for x in domain]
        current_fn = df
    end

    # 2. Lipschitz Constant Estimation: L = max |dK/dζ|
    lipschitz_constant = maximum(abs.(ders[1]))
    max_deriv_magnitudes = [maximum(abs.(ders[k])) for k in 1:max_derivative]

    # 3. Jacobian Condition Number κ(J) across Domain
    cond_numbers = Vector{T}(undef, length(domain))
    J_buffer = zeros(T, length(u0), length(u0))
    for (i, x) in enumerate(domain)
        # Compute Jacobian of PDE residual at state x
        res_at_x(u) = (out = similar(u); residual_fn!(out, u, x); out)
        J = ForwardDiff.jacobian(res_at_x, u0)
        cond_numbers[i] = cond(J)
    end
    max_cond = maximum(filter(x -> !isnan(x) && !isinf(x), cond_numbers))

    # 4. Newton-Raphson Iteration Metrics
    iters_required = Int[]
    failures = 0
    for x in domain
        u = copy(u0)
        res = zeros(T, length(u0))
        converged = false

        for iter in 1:max_iters
            residual_fn!(res, u, x)
            if norm(res, 2) < tol
                push!(iters_required, iter)
                converged = true
                break
            end
            J = ForwardDiff.jacobian(u_vec -> (out = similar(u_vec); residual_fn!(out, u_vec, x); out), u)
            delta = J \ (-res)
            u .+= delta
        end
        if !converged
            failures += 1
            push!(iters_required, max_iters)
        end
    end

    return (
        derivatives = ders,
        max_derivative_magnitudes = max_deriv_magnitudes,
        lipschitz_constant = lipschitz_constant,
        max_jacobian_cond = max_cond,
        mean_newton_iters = mean(iters_required),
        max_newton_iters = maximum(iters_required),
        solver_failure_rate = failures / length(domain)
    )
end

end
```

```julia
module ModalBudgetDiagnostic

using LinearAlgebra

export ModalBudget, compute_modal_tendencies!

struct ModalBudget{T<:Real}
    da_dt::Vector{T}
    F_adv::Vector{T}
    F_diff::Vector{T}
    F_lin::Vector{T}
    F_bc::Vector{T}
    residual::Vector{T}
end

"""
Decomposes Gegenbauer expansion modal tendencies:
da_n/dt = F_adv^(n) + F_diff^(n) + F_lin^(n) + F_bc^(n)
"""
function compute_modal_tendencies!(
    budget::ModalBudget{T},
    a_n::Vector{T},
    F_adv::Vector{T},
    F_diff::Vector{T},
    F_lin::Vector{T},
    tau_s::T,
    q_s::T,
    basis_weights::Vector{T}
) where T<:Real

    num_modes = length(a_n)

    # Project surface stress τ_s and sensible heat flux q_s onto modal basis
    @inbounds for n in 1:num_modes
        budget.F_bc[n] = (tau_s + q_s) * basis_weights[n]
        budget.F_adv[n] = F_adv[n]
        budget.F_diff[n] = F_diff[n]
        budget.F_lin[n] = F_lin[n]

        # Balance equation
        budget.da_dt[n] = budget.F_adv[n] + budget.F_diff[n] + budget.F_lin[n] + budget.F_bc[n]
        budget.residual[n] = budget.da_dt[n] - (budget.F_adv[n] + budget.F_diff[n] + budget.F_lin[n] + budget.F_bc[n])
    end

    @assert maximum(abs.(budget.residual)) < 1e-10 "Conservation budget violation!"
    return budget
end

end
```

```julia
module StatisticalValidation

using LinearAlgebra, Statistics, Random

export ModelValidationMetrics, evaluate_full_statistical_suite

struct ModelValidationMetrics{T<:Real}
    rss::T
    r2::T
    adj_r2::T
    aic::T
    bic::T
    press::T
    rmse_cv::T
    param_ci_lower::Vector{T}
    param_ci_upper::Vector{T}
end

"""
    evaluate_full_statistical_suite(X, y, θ_hat, fit_fn, predict_fn; k_folds=5, n_bootstraps=200)

Calculates Adjusted R², AIC, BIC, PRESS, K-Fold RMSE, and 95% Bootstrap Confidence Intervals.
"""
function evaluate_full_statistical_suite(
    X::Matrix{T},
    y::Vector{T},
    θ_hat::Vector{T},
    fit_fn::Function,
    predict_fn::Function;
    k_folds::Int=5,
    n_bootstraps::Int=200
) where T<:Real

    N, k = size(X)
    y_pred = predict_fn(θ_hat, X)
    residuals = y .- y_pred
    rss = sum(residuals .^ 2)
    tss = sum((y .- mean(y)) .^ 2)

    # Standard R² and Adjusted R²
    r2 = 1.0 - (rss / tss)
    adj_r2 = 1.0 - ((1.0 - r2) * (N - 1) / (N - k - 1))

    # Information Criteria
    rss_safe = max(rss, eps(T))
    aic = 2 * k + N * log(rss_safe / N)
    bic = k * log(N) + N * log(rss_safe / N)

    # PRESS Statistic (Prediction Residual Sum of Squares) via Hat Matrix
    H = X * ((X' * X) \ X')
    press_residuals = residuals ./ (1.0 .- diag(H))
    press = sum(press_residuals .^ 2)

    # K-Fold Cross-Validation Residual Score
    fold_sizes = fill(div(N, k_folds), k_folds)
    fold_sizes[1:mod(N, k_folds)] .+= 1
    indices = randperm(N)
    starts = [1; cumsum(fold_sizes)[1:end-1] .+ 1]
    stops = cumsum(fold_sizes)

    rss_cv = 0.0
    for fold in 1:k_folds
        test_idx = indices[starts[fold]:stops[fold]]
        train_idx = setdiff(indices, test_idx)

        θ_fold = fit_fn(X[train_idx, :], y[train_idx])
        y_cv_pred = predict_fn(θ_fold, X[test_idx, :])
        rss_cv += sum((y[test_idx] .- y_cv_pred) .^ 2)
    end
    rmse_cv = sqrt(rss_cv / N)

    # Bootstrap 95% Confidence Intervals
    boot_params = zeros(T, n_bootstraps, k)
    for b in 1:n_bootstraps
        boot_idx = rand(1:N, N)
        boot_params[b, :] = fit_fn(X[boot_idx, :], y[boot_idx])
    end

    ci_lower = [quantile(boot_params[:, j], 0.025) for j in 1:k]
    ci_upper = [quantile(boot_params[:, j], 0.975) for j in 1:k]

    return ModelValidationMetrics(
        rss, r2, adj_r2, aic, bic, press, rmse_cv, ci_lower, ci_upper
    )
end

end
```

```julia
Hierarchical Bayesian model pooling hyperpriors across field sites:
Global Atmos -> Site Realization -> Observed Vertical Profiles
"""
@model function hierarchical_closure_model(
    data_by_site::Vector{Tuple{Matrix{Float64}, Vector{Float64}}},
    num_params::Int
)

    # 1. Global Hyperpriors (Universal Physics Constants)
    μ_global ~ MvNormal(zeros(num_params), 10.0 * I)
    σ_global ~ MvNormal(ones(num_params), 2.0 * I)

    # Positive variance bounds
    σ_bounded = abs.(σ_global) .+ 1e-4
    Ω = Diagonal(σ_bounded .^ 2)

    # 2. Site-Specific Parameter Deviations
    θ_site = Vector{Vector{Real}}(undef, num_sites)
    for i in 1:num_sites
        θ_site[i] ~ MvNormal(μ_global, Ω)
    end

    # 3. Observation Likelihood per Site
    for i in 1:num_sites
        X_site, y_site = data_by_site[i]
        σ_obs ~ Exponential(1.0)

        y_hat = X_site * θ_site[i]
        y_site ~ MvNormal(y_hat, (σ_obs^2) * I)
    end
end
```

## Mathematical Formulation & Desingularization

In fast-slow atmospheric formulations where fast variables $\mathbf{y} \in \mathbb{R}^m$ (e.g., turbulent kinetic energy $e$, Reynolds stresses) and slow variables $\mathbf{z} \in \mathbb{R}^n$ (e.g., mean velocity profile $u(z)$, potential temperature $\theta(z)$) evolve as:

$$\epsilon \frac{d\mathbf{y}}{dt} = \mathbf{g}(\mathbf{y}, \mathbf{z}), \qquad \frac{d\mathbf{z}}{dt} = \mathbf{f}(\mathbf{y}, \mathbf{z}), \qquad 0 < \epsilon \ll 1$$

The critical manifold $\mathcal{S}_0$ is defined by $\mathbf{g}(\mathbf{y}, \mathbf{z}) = \mathbf{0}$. The fold set $\mathcal{F} \subset \mathcal{S}_0$ marks the boundary where normal hyperbolicity breaks down:

$$\mathcal{F} = \left\{ (\mathbf{y}, \mathbf{z}) \in \mathcal{S}_0 \,\,\middle\vert{}\,\, \det \mathbf{D}_{\mathbf{y}}\mathbf{g}(\mathbf{y}, \mathbf{z}) = 0 \right\}$$

Differentiating $\mathbf{g}(\mathbf{y}(\mathbf{z}), \mathbf{z}) = \mathbf{0}$ with respect to slow time $t$ yields the constrained slow flow:

$$\mathbf{D}_{\mathbf{y}}\mathbf{g} \cdot \frac{d\mathbf{y}}{dt} + \mathbf{D}_{\mathbf{z}}\mathbf{g} \cdot \mathbf{f}(\mathbf{y}, \mathbf{z}) = \mathbf{0} \implies \det(\mathbf{D}_{\mathbf{y}}\mathbf{g}) \frac{d\mathbf{y}}{dt} = -\operatorname{adj}(\mathbf{D}_{\mathbf{y}}\mathbf{g}) \mathbf{D}_{\mathbf{z}}\mathbf{g} \cdot \mathbf{f}(\mathbf{y}, \mathbf{z})$$

Rescaling time via $d\tau = \frac{dt}{\det(\mathbf{D}_{\mathbf{y}}\mathbf{g})}$ yields the **desingularized slow flow**:

$$\frac{d\mathbf{y}}{d\tau} = -\operatorname{adj}(\mathbf{D}_{\mathbf{y}}\mathbf{g}) \mathbf{D}_{\mathbf{z}}\mathbf{g} \cdot \mathbf{f}(\mathbf{y}, \mathbf{z})$$

Folded singularities $(\mathbf{y}^*, \mathbf{z}^*) \in \mathcal{F}$ satisfy:

$$\operatorname{adj}(\mathbf{D}_{\mathbf{y}}\mathbf{g}(\mathbf{y}^*, \mathbf{z}^*)) \, \mathbf{D}_{\mathbf{z}}\mathbf{g}(\mathbf{y}^*, \mathbf{z}^*) \cdot \mathbf{f}(\mathbf{y}^*, \mathbf{z}^*) = \mathbf{0}$$

### Classification Criteria

Evaluating the eigenvalues $\lambda_1, \lambda_2$ of the desingularized Jacobian Matrix $\mathbf{J}_{\text{desing}} = \nabla_{\mathbf{y},\mathbf{z}} \left( \frac{d\mathbf{y}}{d\tau} \right)$ at $(\mathbf{y}^*, \mathbf{z}^*)$:

* **Folded Saddle ($\lambda_1 \cdot \lambda_2 < 0, \, \lambda_i \in \mathbb{R}$):** Supports singular canards (True Canards). Two unique trajectories cross from the attracting manifold $\mathcal{S}_0^a$ to the repelling manifold $\mathcal{S}_0^r$.
* **Folded Node ($\lambda_1 \cdot \lambda_2 > 0, \, \lambda_i \in \mathbb{R}$):** Supports a Canard Funnel. A continuous wedge of trajectories passes smoothly across the fold onto $\mathcal{S}_0^r$.
* **Folded Focus ($\text{Im}(\lambda_{1,2}) \neq 0$):** No smooth crossing possible. Trajectories undergo immediate fast jump detachment.

## Executable Detection Engine (`src/Geometry/CanardDetection.jl`)

```julia
module CanardDetection

using LinearAlgebra, ForwardDiff, DifferentialEquations

export FoldedSingularity, classify_folded_singularity, detect_canard_trajectories

@enum SingularityType FoldedSaddle FoldedNode FoldedFocus NonSingular

struct FoldedSingularity{T<:Real}
    state_y::Vector{T}
    state_z::Vector{T}
    type::SingularityType
    eigenvalues::Vector{Complex{T}}
    ratio_μ::T
end

"""
    desingularized_rhs(y, z, g_fn, f_fn)

Computes the right-hand side of the desingularized slow flow:
  dy/dτ = - adj(D_y g) * D_z g * f(y, z)
"""
function desingularized_rhs(y::AbstractVector{T}, z::AbstractVector{T}, g_fn::Function, f_fn::Function) where T<:Real
    J_y = ForwardDiff.jacobian(y_vec -> g_fn(y_vec, z), y)
    J_z = ForwardDiff.jacobian(z_vec -> g_fn(y, z_vec), z)
    f_val = f_fn(y, z)

    # Calculate adjugate matrix adj(A) = det(A) * inv(A)
    adj_J_y = det(J_y) * inv(J_y)

    return -adj_J_y * J_z * f_val
end

"""
    classify_folded_singularity(y_star, z_star, g_fn, f_fn)

Evaluates the desingularized Jacobian at a fold point to classify it as a
Folded Saddle, Folded Node, or Folded Focus.
"""
function classify_folded_singularity(
    y_star::Vector{T},
    z_star::Vector{T},
    g_fn::Function,
    f_fn::Function
) where T<:Real

    # Define state vector x = [y; z]
    x_star = [y_star; z_star]
    ny = length(y_star)

    rhs_full(x) = begin
        y_loc = x[1:ny]
        z_loc = x[(ny+1):end]
        desingularized_rhs(y_loc, z_loc, g_fn, f_fn)
    end

    J_desing = ForwardDiff.jacobian(rhs_full, x_star)
    eigs = eigen(J_desing).values

    # Non-zero eigenvalues relevant to slow manifold dynamics
    non_zero_eigs = filter(e -> abs(e) > 1e-7, eigs)

    if length(non_zero_eigs) < 2
        return FoldedSingularity(y_star, z_star, NonSingular, eigs, zero(T))
    end

    λ1, λ2 = non_zero_eigs[1], non_zero_eigs[2]
    ratio = Real(λ1 / λ2)

    stype = if imag(λ1) != 0 || imag(λ2) != 0
        FoldedFocus
    elseif real(λ1) * real(λ2) < 0
        FoldedSaddle
    else
        FoldedNode
    end

    return FoldedSingularity(y_star, z_star, stype, eigs, ratio)
end

"""
    detect_canard_trajectories(sol, g_fn, fold_threshold=1e-3)

Scans an integrated trajectory sol(t) for persistent movement along repelling
sheets S_0^r past fold points det(D_y g) = 0.
"""
function detect_canard_trajectories(sol::RODESolution, g_fn::Function; fold_threshold::Float64=1e-3)
    canard_segments = Tuple{Float64, Float64}[]
    in_repelling_regime = false
    t_entry = 0.0

    for i in 1:length(sol.t)
        u_t = sol.u[i]
        # Split into fast y and slow z components based on domain size
        ny = div(length(u_t), 2)
        y, z = u_t[1:ny], u_t[(ny+1):end]

        J_y = ForwardDiff.jacobian(y_vec -> g_fn(y_vec, z), y)
        det_J = det(J_y)

        # det(D_y g) > 0: Attracting Sheet S_0^a; det(D_y g) < 0: Repelling Sheet S_0^r
        if det_J < -fold_threshold && !in_repelling_regime
            in_repelling_regime = true
            t_entry = sol.t[i]
        elseif det_J >= -fold_threshold && in_repelling_regime
            in_repelling_regime = false
            t_exit = sol.t[i]
            # Record if repelling duration persistence is O(1) in slow time
            if (t_exit - t_entry) > 0.1
                push!(canard_segments, (t_entry, t_exit))
            end
        end
    end

    return canard_segments
end

end
```
